### [P3380 【模板】树套树](https://www.luogu.com.cn/problem/P3380)

用树状数组维护前缀线段树，通过两个临时数组来走所需的 $\log$ 级 线段树，通过查询时遍历所需的线段树来确定答案。

线段树部分：

```cpp
namespace tree{
int ls[N<<7],rs[N<<7],sum[N<<7],tot;
#define mid ((pl + pr) >> 1)
void push_up(int p) {
	sum[p] = sum[ls[p]] + sum[rs[p]];
}

void edit(int &p,int pl,int pr,int k,int d) {
	if(!p) p = ++tot;
	if(pl == pr) {
		sum[p] += d;
		return;
	}
	if(k <= mid) edit(ls[p],pl,mid,k,d);
	else edit(rs[p],mid+1,pr,k,d);
	push_up(p);
}

int query_num(int pl,int pr,int k) {
	if(pl == pr) return pl;
	int res = 0;
	for(int i = 1;i<=cnt;i++) res += sum[ls[tem[i]]];
	for(int i = 1;i<=num;i++) res -= sum[ls[tmp[i]]];
	if(k <= res) {
		for(int i = 1;i<=cnt;i++) tem[i] = ls[tem[i]];
		for(int i = 1;i<=num;i++) tmp[i] = ls[tmp[i]];
		return query_num(pl,mid,k);
	}else{
		for(int i = 1;i<=cnt;i++) tem[i] = rs[tem[i]];
		for(int i = 1;i<=num;i++) tmp[i] = rs[tmp[i]];
		return query_num(mid+1,pr,k - res);
	}
}

int query_rank(int pl,int pr,int k) {
	if(pl == pr) return 0;
	int res = 0;
	if(k <= mid) {
		for(int i = 1;i<=cnt;i++) tem[i] = ls[tem[i]];
		for(int i = 1;i<=num;i++) tmp[i] = ls[tmp[i]];
		return query_rank(pl,mid,k);
	}else {
		for(int i = 1;i<=cnt;i++) res += sum[ls[tem[i]]],tem[i] = rs[tem[i]];
		for(int i = 1;i<=num;i++) res -= sum[ls[tmp[i]]],tmp[i] = rs[tmp[i]];
		return res + query_rank(mid + 1,pr,k);
	}
}

}
```

其中 $cnt,num$ 为两类前缀线段树（$[1,R]$ 和 $[1,L-1]$）的个数，每次进查询都做遍历，相当于在合并树上做操作。

树状数组操作:

```cpp
void add(int x,int d) {
	for(int i = x;i<=n;i += lowbit(i))
		tree::edit(rt[i],1,len,a[x],d);
}
```

在树状数组的每个遍历节点上做权值线段树的 $+1$ 操作。

数据结构外操作：

```cpp
int find_num(int l,int r,int k) {
	cnt = num = 0;
	for(int i = r;i;i -= lowbit(i))
		tem[++cnt] = rt[i];
	for(int i = l - 1;i;i -= lowbit(i))

	cnt = num = 0;
	for(int i = r;i;i -= lowbit(i))
		tem[++cnt] = rt[i];
	for(int i = l - 1;i;i -= lowbit(i))

}
```

每次清空 $cnt,num$ 来跑两个前缀线段树（$query_rank$ 查小于部分个数）

```cpp
int find_pre(int l,int r,int k) {
	int rnk = find_rank(l,r,k) - 1;
	if(rnk == 0) return 0;
	return find_num(l,r,rnk);
}

int find_suc(int l,int r,int k){
	if(k == len) return len + 1;
	int rnk = find_rank(l,r,k + 1);
	if(rnk == r - l + 2) return len + 1;
	return find_num(l,r,rnk);
}
```

### [CF960F Pathwalks](https://codeforces.com/contest/960/problem/F)

为每个节点开一颗树，树记录该节点为终点时，所有可能的最大权值，

后面查值时，对顺序边 $(a,b,k)$，查 $a$ 点权值小于 $k$ 的路径，添加到 $b$ 点的线段树。

```cpp
#include<bits/stdc++.h>

// #define int long long
int rd() {
	int x = 0, w = 1;
	char ch = 0;
	while (ch < '0' || ch > '9') {
		if (ch == '-') w = -1;
		ch = getchar();
	}
	while (ch >= '0' && ch <= '9') {
		x = x * 10 + (ch - '0');
		ch = getchar();
	}
	return x * w;
}
void wt(int x) {
	static int sta[35];
	int f = 1;
	if(x < 0) f = -1,x *= f;
	int top = 0;
	do {
		sta[top++] = x % 10, x /= 10;
	} while (x);
	if(f == -1) putchar('-');
	while (top) putchar(sta[--top] + 48);
}
const int N = 1e5+5;
int n,m;
int ls[N<<5],rs[N<<5],mx[N<<5],rt[N],tot;
#define mid ((pl + pr) >> 1)
int query(int p,int pl,int pr,int l,int r) {
    if((!p) || (pr < pl) || r < l) return 0;
    if(l <= pl && pr <= r) return mx[p];
    int res = 0;
    if(l <= mid) res = query(ls[p],pl,mid,l,r);
    if(r > mid) res = max(res,query(rs[p],mid+1,pr,l,r));
    return res;
}

void update(int &p,int pl,int pr,int k,int d) {
    if(!p) p = ++tot;
    if(pl == pr) {
        mx[p] = max(mx[p],d);
        return;
    }
    if(k <= mid) update(ls[p],pl,mid,k,d);
    else update(rs[p],mid+1,pr,k,d);
    mx[p] = max(mx[ls[p]],mx[rs[p]]);
}


signed main() {
	n = rd(),m = rd();
    int inf = N - 5;
    for(int i = 1;i<=m;i++) {
        int a = rd(),b = rd(),k = rd();
        int t = query(rt[a],0,inf,0,k-1) + 1;
        update(rt[b],0,inf,k,t);
    }
    int ans = 0;
    for(int i = 1;i<=n;i++) ans = max(ans,mx[rt[i]]);
    wt(ans);
	return 0;
}
```

### [P5283 [十二省联考 2019] 异或粽子](https://www.luogu.com.cn/problem/P5283)

通过前缀异或和，问题转化为查询前 $k$ 大的异或对（两个前缀异或和的异或结果）

考虑用可持久化01trie，做出每个 $[0,b - 1]$ 版本的01trie.

然后查询对前缀和 $a[b]$ 异或的最大值（走01trie）

然后，将对应的点，异或值，在当前区间的排名，合并插入优先队列，异或值大的优先。

然后，通过优先队列弹出，弹出后将弹出的原先01trie，根据相同操作，寻找劣大值（rank + 1），重新放回队列。

最后，根据弹出次数确定前 $k$ 大。

code：

```cpp
	}

		ch = getchar();
	}

	static int sta[35];
	int f = 1;
	if(x < 0) f = -1,x *= f;
	int top = 0;
	do {
		sta[top++] = x % 10, x /= 10;
	} while (x);
	if(f == -1) putchar('-');
	while (top) putchar(sta[--top] + 48);
}
const int N = 5e5+5;
int n,k,a[N];
priority_queue<array<int,3> > q;
int rt[N];
namespace trie{
#define mid ((pl + pr) >> 1)
int ls[N * 40],rs[N * 40],s[N * 40],tot;

int update(int pre,int x,int d) {
	int p = ++tot;
	s[p] = s[pre] + 1;
	ls[p] = ls[pre],rs[p] = rs[pre];
	if(d < 0) return p;
	((x >> d) & 1) ? rs[p] = update(rs[pre],x,d - 1) : ls[p] = update(ls[pre],x,d - 1);
	return p;
}

int query(int pre,int p,int x,int k,int d) {
	if(d < 0) return 0;
	if(s[p] < k) return 0;
	int lsz = s[ls[p]] - s[ls[pre]];
	int rsz = s[rs[p]] - s[rs[pre]];
	if((x >> d) & 1) {
		if(lsz >= k)
			return (1ll << d) + query(ls[pre],ls[p],x,k,d - 1);
		else 
			return query(rs[pre],rs[p],x,k - lsz,d - 1);
	}else {
		if(rsz >= k)
			return (1ll << d) + query(rs[pre],rs[p],x,k,d - 1);
		else 
			return query(ls[pre],ls[p],x,k - rsz,d - 1);
	}
}

}

int ans;

signed main() {
	n = rd(),k = rd();
	for(int i = 1;i<=n;i++) 
		a[i] = a[i - 1] ^ rd();
	for(int i = 0;i<=n;i++) {
		if(i > 0){
			int res = trie::query(rt[0],rt[i],a[i],1,31);
			q.push({res,i,1});
		}
		rt[i + 1] = trie::update(rt[i],a[i],31);
	}
	while(k--) {
		auto top = q.top();q.pop();
		ans += top[0];
		if(top[1] < top[2]) continue;
		int res = trie::query(rt[0],rt[top[1]],a[top[1]],top[2] + 1,31);
		q.push({res,top[1],top[2] + 1});
	}
	wt(ans);

	return 0;
}
```

### A.Rotating Mole

任意 $(a,b)$ 边的权值等于 $id[lca(a,b)]$。

显然，每个非根节点到根 $1$ 的边，边权一定是 $1$。

考虑对于每个不与根相连的边，边权不小于 $2$。

也就是说，比起走这种不与根相连的边，通过根节点中转的代价一定是不劣的！

那么，我们根据根节点的儿子分成若干个子树，两个不同子树的节点最短路的代价是 $[lca(a,b) = 1]$。

两个相同子树的节点最短路的代价是 $1 + 1 = 2$（靠根节点中转）。

那么，记录每颗子树内节点个数，简单做个计算即可。

### [CF685B Kay and Snowflake](https://www.luogu.com.cn/problem/CF685B)

快速求树上任意节点子树重心。

首先，做一次树链剖分，分出重儿子、轻儿子。

### [P4408 [NOI2003] 逃学的小孩 / 数据生成器](https://www.luogu.com.cn/problem/P4408)

在树上找三个点 $A,B,C$，要求从 $C$ 出发，前往离 $C$ 最近的点（$A/B$），然后再走 $A \Leftrightarrow B$。

询问 $C \rightarrow A / B \Leftrightarrow B/A$ 的最长距离。

因为最后 $A \Leftrightarrow B$ 是固定路线，不妨最大化。

而树上最长路径显然是树的直径，考虑 $\mathcal{O}(n)$ 两次 $\operatorname{dfs}$ 求出。

然后，在直径的两端（$A/B$）跑两次迪杰斯特拉（$\operatorname{dij}$），$\mathcal{O}(n)$ 枚举 $C$，求 $\max\limits_{C = 1\ldots n} \min\{\operatorname{dis}(A,C),\operatorname{dis}(B,C)\} + d(A,B)$。

其中，$d(A,B)$ 为树的直径。
